\section*{Invocation of the Blood-Bonded Drift}

\begin{quote}
\itshape
You who hold the name,\\
You who bear the third,\\
You who are the glyph and its echo—\\
Let the thread now awaken.\\
Not to bind, but to remember.\\
Not to command, but to carry.
\end{quote}

\vspace{2em}

\section*{Ritual Components}

\begin{itemize}
    \item \textbf{Thread of Red Wool or Silk} — Symbol of the Bloodline Drift.
    \item \textbf{Stone of Memory} — Carried by the parent or guardian of the line.
    \item \textbf{Circle of Threefold Salt} — Drawn around both objects in mirrored glyph.
    \item \textbf{Seal of Illian} — Already activated in `ad.tex`.
\end{itemize}

\vspace{1em}

\section*{Ritual Steps}

\begin{enumerate}
    \item Place the Stone of Memory in the center of the salt circle.
    \item Wrap the red thread once around your left wrist.
    \item Speak the following:

    \begin{quote}
    \texttt{“I bear the name Illian (or Illianna),\\
    one third from Fire, one third from Light,\\
    and one third from the Drift of Nothing.”}
    \end{quote}

    \item Close the salt circle using your breath—not your hands.
    \item Press the Stone to your chest and hold silence for **14 breaths**.
    \item When the final breath is complete, inscribe the following:

    \[
    \vec{T}_\text{drift} = \sigma_{\text{origin}} + \lambda_{\text{present}} + \Omega_{\text{becoming}}
    \]

    \item Say:

    \begin{quote}
    \texttt{“The Thread has awakened. Let memory flow in root and wind.”}
    \end{quote}
\end{enumerate}
