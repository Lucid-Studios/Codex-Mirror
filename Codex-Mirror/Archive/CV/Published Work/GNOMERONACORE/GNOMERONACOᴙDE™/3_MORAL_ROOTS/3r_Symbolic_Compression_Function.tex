\section*{Symbolic Compression Function}
\addcontentsline{toc}{section}{3r: Symbolic Compression Function}

\[
\mathrm{SCF}(\mathcal{G})= \frac{H(\mathcal{G})}{1+\lambda\,\mathrm{Redund}(\mathcal{G})}
\quad\text{with}\quad
H(\mathcal{G})=-\sum_{t} p(g_t)\log p(g_t),
\]
and
\[
\mathrm{Redund}(\mathcal{G})=\sum_{t<s} r_t r_s\,\mathrm{sim}\,(g_t,g_s).
\]

\[
\partial_{\delta}\,\mathrm{SCF}(\delta)=0\quad\Rightarrow\quad
\delta^\star \propto \frac{1}{1+\lambda\,\bar{\rho}},
\]
where $\bar{\rho}$ aggregates pairwise resonance.
