\[
M_{\text{total}}=(M_{\text{myth}}\otimes M_{\text{deck}})\;\oplus\; (M_{\text{myth}}\,P\,M_{\text{nlp}}^\top),
\]

\[
x^{(t+1)}=\sigma\!\bigl(M_{\text{total}}\,x^{(t)}\bigr),
\]
