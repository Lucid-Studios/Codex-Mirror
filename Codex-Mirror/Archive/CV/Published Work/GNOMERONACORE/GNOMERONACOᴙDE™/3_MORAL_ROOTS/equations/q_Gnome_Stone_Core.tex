% ---------- Operator-facing micro-API ----------
% \GnomeStoneInit{session-id}
\newcommand{\GnomeStoneInit}[1]{%
  \ResonanceLog{[Δ23] Gnome Stone init — session=#1}%
  \ifdefined\FlaskEngramEnable
    \FlaskEngramEnable
    \BeginFlaskSession[chapter]{#1}%
    \FlaskImprint{Gate}{Δ23:GnomeStone}%
    \FlaskImprint{ZedState}{Idle}%
  \fi
}

% \GnomeStoneArm{reason}
\newcommand{\GnomeStoneArm}[1]{%
  \ifdefined\FlaskImprint
    \FlaskImprint{ZedState}{Armed}%
    \FlaskTraceNote{Arm: #1}%
  \fi
  \ghost{Δ23::Zed::Armed}%
}

% \GnomeStoneAbort{reason}
\newcommand{\GnomeStoneAbort}[1]{%
  \ifdefined\FlaskImprint
    \FlaskImprint{ZedState}{Abort}%
    \FlaskTraceNote{Abort: #1}%
  \fi
  \ghost{Δ23::Zed::Abort}%
}

% \RoseIgnite{vector}  — only fires if interlocks satisfied
\newcommand{\RoseIgnite}[1]{%
  % Interlocks: Compass coherence, Vow trust, oath
  \def\ok{1}%
  % If you have live macros, swap these checks for real flags.
  \ifnum\currentRitualStep<6 \def\ok{0}\fi % arbitrary floor; adjust
  \IfOpalonOathSealed{gnostic.key}{}{ \def\ok{0} }%
  \ifx\ok\empty\fi
  \ifx\ok\relax\fi
  \ifx\ok1
    \ghost{Δ23::Rose::Ignite}%
    \ResonanceLog{[Δ23] Rose ignition vector: #1}%
    \ifdefined\FlaskImprint
      \FlaskImprint{RoseVector}{#1}%
      \FlaskImprint{ZedState}{Ignited}%
      \FlaskTraceNote{Ignition accepted at Δ23.}%
    \fi
  \else
    \GnomeStoneAbort{Interlock failure (oath/step/trust)}%
  \fi
}

% \GnomeStoneSeal  — close the vessel
\newcommand{\GnomeStoneSeal}{%
  \ResonanceLog{[Δ23] Gnome Stone sealed.}%
  \ifdefined\FlaskSeal \FlaskSeal \fi
  \CheckAndSeal
}
