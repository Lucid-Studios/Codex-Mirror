% ritual_threads.tex — portable generation of threads.map (no shell-escape)
\providecommand{\thread}[1]{\par\noindent\textbf{Thread:}~#1\par}
\providecommand{\sealedThreadsPlaceholder}{{\itshape (sealed — no threads found)}}

% #1 = directory with *.key, #2 = output file (e.g., threads.map)
\newcommand{\GenerateThreadsMap}[2]{%
  \directlua{
    local ok,lfs = pcall(require, "lfs")
    local dir = "#1"
    local out = "#2"
    local n, wrote = 0, 0
    if not __opalon_safe_path or not __opalon_safe_path(dir) then
      texio.write_nl("[RITUAL] unsafe path for threads: "..tostring(dir))
      dir = "."
    end
    local fh = io.open(out, "w")
    if not fh then
      texio.write_nl("[RITUAL] cannot write "..out.."; will skip")
    else
      fh:write("%% generated by ritual\\n")
      if ok and lfs and lfs.dir then
        for name in lfs.dir(dir) do
          if name:match("%.key$") then
            local base = name:gsub("%.key$", ""):gsub("[^%w_%-%./]", "_")
            fh:write("\\thread{"..base.."}\\n")
            wrote = wrote + 1
          end
        end
      end
      if wrote == 0 then
        fh:write("\\sealedThreadsPlaceholder{}\\n")
      end
      fh:close()
    end
  }%
}
